/* Orivo theme toggle — injects a sun/moon button into .nav and persists choice. */ (function(){ var KEY="orivo-theme"; function apply(t){ document.documentElement.setAttribute("data-theme",t); var b=document.querySelector(".theme-toggle"); if(b){b.textContent=t==="light"?"\u263E":"\u2600"; b.setAttribute("aria-label",t==="light"?"Switch to dark mode":"Switch to light mode");} } var saved=null; try{saved=localStorage.getItem(KEY);}catch(e){} var theme=saved||"dark"; document.documentElement.setAttribute("data-theme",theme); function init(){ var nav=document.querySelector(".nav"); if(!nav)return; var btn=document.createElement("button"); btn.className="theme-toggle"; btn.type="button"; btn.addEventListener("click",function(){ theme=(document.documentElement.getAttribute("data-theme")==="light")?"dark":"light"; try{localStorage.setItem(KEY,theme);}catch(e){} apply(theme); }); var links=nav.querySelector(".nav-links"); if(links){links.appendChild(btn);}else{nav.appendChild(btn);} apply(theme); } if(document.readyState==="loading"){document.addEventListener("DOMContentLoaded",init);}else{init();} })();
